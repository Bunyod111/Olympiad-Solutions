    // === ФУНКЦИЯ ОТПРАВКИ С АВТО-ПОВТОРОМ (ДЛЯ 700 ЧЕЛОВЕК) ===
    async function sendDataWithRetry(data, retries = 4) {
        for (let i = 0; i < retries; i++) {
            try {
                // Если запрос висит дольше 15 сек — сбрасываем
                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), 15000);

                const response = await fetch(GOOGLE_SCRIPT_URL, {
                    method: 'POST',
                    body: JSON.stringify(data),
                    signal: controller.signal
                });
                clearTimeout(timeoutId);

                const result = await response.json();

                if (result.status === "error" && result.message === "Server busy") {
                    throw new Error("Server busy"); 
                }
                
                return result; 
            } catch (err) {
                console.warn(`Попытка ${i + 1} не удалась: ${err.message}`);
                
                if (i === retries - 1) throw err;
                
                const waitTime = 1500 + Math.random() * 1000;
                await new Promise(res => setTimeout(res, waitTime));
            }
        }
    }

            alert("Вопросы для этого класса еще не загружены.");
            submitButton.disabled = false;
            submitButton.textContent = uiText[currentLang].regBtn;
            return;
        }

        sendDataWithRetry({
            type: 'start_session',
            phone: currentParticipant.phone
        })
        .then(serverData => {
            if (serverData.status === "error") {
                if (serverData.message === "ALREADY_SUBMITTED") {
                    alert(uiText[currentLang].alreadyDone);
                } else {
                    alert("Error: " + serverData.message);
                }
                submitButton.disabled = false;
                submitButton.textContent = uiText[currentLang].regBtn;
                return;
            }

            const remainingSeconds = serverData.timeLeft;

            if (remainingSeconds <= 0) {
                alert(uiText[currentLang].timeOut);
                return;
            }

            if (remainingSeconds < 90 * 60) {
                alert(uiText[currentLang].rejoin);
            }

            loadQuiz(selectedGrade);
            startTimer(remainingSeconds); 

            regSection.classList.remove("active-section");
            regSection.classList.add("hidden-section");
            document.querySelector('.lang-switcher').style.display = 'none';

            submitButton.textContent = uiText[currentLang].regBtn;
        });
    });

    // === ЗАГРУЗКА ТЕСТА ===
    function loadQuiz(grade) {
        quizHeader.textContent = `${uiText[currentLang].quizHeader} ${grade} ${uiText[currentLang].classLabel}`;
        const questions = questionsDB[currentLang][grade] || [];
        quizForm.innerHTML = ""; 

        questions.forEach((q, index) => {
            const questionBlock = document.createElement("div");
            questionBlock.className = "question-block";
            
            const questionText = document.createElement("p");
            questionText.innerHTML = `${index + 1}. ${q.q}`; 
            questionBlock.appendChild(questionText);

            q.options.forEach(option => {
                const label = document.createElement("label");
                const radio = document.createElement("input");
                radio.type = "radio";
                radio.name = `q-${index}`;
                radio.value = option; 
                
                label.appendChild(radio);
                label.append(` ${option}`); 
                questionBlock.appendChild(label);
            });
            quizForm.appendChild(questionBlock);
        });
    }

        clearInterval(timerInterval); 

        quizSubmitButton.disabled = true;
        quizSubmitButton.textContent = uiText[currentLang].waitText;

        const userAnswers = [];
        const questionsCount = questionsDB[currentLang][selectedGrade].length;
        const formData = new FormData(quizForm);

        for (let i = 0; i < questionsCount; i++) {
            const ans = formData.get(`q-${i}`);
            userAnswers.push(ans ? ans : ""); 
        }

        const dataToSend = {
            type: 'submit_quiz',
            ...currentParticipant,
            userAnswers: userAnswers
        };

        sendDataWithRetry(dataToSend)
        .then(serverData => {
            if (serverData.status === "error") {
                alert("Server Error: " + serverData.message);
                quizSubmitButton.disabled = false;
                quizSubmitButton.textContent = uiText[currentLang].submitQuiz;
                return;
            }
            showResult(serverData.score, serverData.percentage, serverData.passStatus, serverData.total);
        })
        .catch(err => {
            console.error("Ошибка сети:", err);
            alert(uiText[currentLang].errorNetwork + "\nПожалуйста, нажмите кнопку еще раз.");
            quizSubmitButton.disabled = false;
            quizSubmitButton.textContent = uiText[currentLang].submitQuiz;
        });
    });

        }
        resultMessage.appendChild(pDetail);

        // 2. Единое сообщение для всех (без поздравлений или сожалений)
        const pFinal = document.createElement("p");
        pFinal.textContent = uiText[currentLang].finalMessage;
        pFinal.style.fontSize = "1.1rem";
        pFinal.style.margin = "20px 0";

    function startTimer(durationInSeconds) {
        quizEndTime = Date.now() + durationInSeconds * 1000; 

        timerDisplay.style.display = "block";
        timerInterval = setInterval(updateTimerDisplay, 1000);
        updateTimerDisplay(); 
    }

    function updateTimerDisplay() {
        const remainingMs = quizEndTime - Date.now();
        const remainingSeconds = Math.round(remainingMs / 1000);

        if (remainingSeconds <= 0) {
            clearInterval(timerInterval); 
            timerDisplay.textContent = "00:00";
            alert(uiText[currentLang].timeOut);
            
            const submitBtn = quizSection.querySelector("button[type='submit']");
            if (submitBtn && !submitBtn.disabled) {
                submitBtn.click();
            }
            return;
        }

        const minutes = Math.floor(remainingSeconds / 60);
        const seconds = remainingSeconds % 60;
        const formattedMinutes = String(minutes).padStart(2, '0');
        const formattedSeconds = String(seconds).padStart(2, '0');

        timerDisplay.textContent = `${uiText[currentLang].timeLeft} ${formattedMinutes}:${formattedSeconds}`;
        
        if (remainingSeconds < 300) { 
             timerDisplay.style.color = "red";
        }
    }
